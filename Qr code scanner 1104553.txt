// 1. Local IndexedDB Storage
// -------------------------------------------------------------
const DB_NAME = 'QRAuditDB';
const DB_VERSION = 1;
const STORE_NAME = 'serials';
let db = null;

function initIndexedDB() {
return new Promise((resolve, reject) => {
const request = indexedDB.open(DB_NAME, DB_VERSION);
request.onupgradeneeded = (event) => {
const dbInstance = event.target.result;
if (!dbInstance.objectStoreNames.contains(STORE_NAME)) {
const store = dbInstance.createObjectStore(STORE_NAME, { keyPath: 'serial' });
store.createIndex('status', 'status', { unique: false });
store.createIndex('synced', 'synced', { unique: false });
}
};
request.onsuccess = (event) => {
db = event.target.result;
renderFromDB();
resolve(db);
};
request.onerror = (e) => reject(e.target.error);
});
}

function saveRecord(record) {
return new Promise((resolve, reject) => {
if (!db) return resolve();
const transaction = db.transaction([STORE_NAME], 'readwrite');
const store = transaction.objectStore(STORE_NAME);
const request = store.put(record);
request.onsuccess = () => { renderFromDB(); resolve(); };
request.onerror = (e) => reject(e.target.error);
});
}

function getAllRecords() {
return new Promise((resolve, reject) => {
if (!db) return resolve([]);
const transaction = db.transaction([STORE_NAME], 'readonly');
const store = transaction.objectStore(STORE_NAME);
const request = store.getAll();
request.onsuccess = () => resolve(request.result);
request.onerror = (e) => reject(e.target.error);
});
}

async function renderFromDB() {
if (!db) return;
const records = await getAllRecords();
const tbody = document.getElementById('dbTableBody');
const recordCount = document.getElementById('recordCount');

recordCount.textContent = records.length + ' records';
if (records.length === 0) {
tbody.innerHTML = '<tr><td colspan="4" style="text-align: center; color: #64748b; padding: 1.5rem; font-style: italic;">No records stored in IndexedDB.</td></tr>';
return;
}

let html = '';
for (let i = 0; i < records.length; i++) {
const r = records[i];
const isVerified = r.status === 'verified';
const badgeClass = isVerified ? 'badge-verified' : 'badge-pending';
const syncColor = r.synced ? '#4ade80' : '#64748b';
const syncText = r.synced ? 'Synced' : 'Pending Sync';
const timeStr = r.timestamp || '-';

html += '<tr>' +
'<td style="font-family: monospace; font-weight: 500; color: #e2e8f0;">' + r.serial + '</td>' +
'<td><span class="badge ' + badgeClass + '">' + r.status + '</span></td>' +
'<td style="font-family: monospace; font-size: 0.625rem; color: ' + syncColor + ';">' + syncText + '</td>' +
'<td style="text-align: right; font-family: monospace; color: #64748b; font-size: 0.625rem;">' + timeStr + '</td>' +
'</tr>';
}
tbody.innerHTML = html;
}

async function processCode(scannedSerial) {
if (!scannedSerial) return;
const rawPayload = scannedSerial.trim();
const display = document.getElementById('latestScanDisplay');

if (!db) return;
const records = await getAllRecords();

let match = records.find(r => r.serial.toLowerCase() === rawPayload.toLowerCase());
if (!match) {
match = records.find(r => r.serial.length >= 4 && rawPayload.toLowerCase().includes(r.serial.toLowerCase()));
}
if (!match) {
match = records.find(r => rawPayload.length >= 4 && r.serial.toLowerCase().includes(rawPayload.toLowerCase()));
}

if (match) {
if (match.status === 'verified') {
display.textContent = 'Already verified: ' + match.serial;
display.style.color = '#fbbf24';
} else {
match.status = 'verified';
match.timestamp = new Date().toLocaleTimeString();
match.synced = false;
await saveRecord(match);

display.textContent = 'VERIFIED MATCH: ' + match.serial;
display.style.color = '#4ade80';
}
} else {
display.textContent = 'UNLISTED SERIAL: ' + rawPayload;
display.style.color = '#f87171';
}
}

// 3. Computer Vision & QR Pipeline Logic (jsQR Integration)
// -------------------------------------------------------------
function decodeImageDataWithCV(imageData) {
if (typeof jsQR === 'undefined') return null;

// Pass 1: Direct Grid Sampling & Decodes
let code = jsQR(imageData.data, imageData.width, imageData.height, {
inversionAttempts: "dontInvert",
});
if (code && code.data) return code.data;

// Pass 2: Inverted Colors
code = jsQR(imageData.data, imageData.width, imageData.height, {
inversionAttempts: "onlyInvert",
});
if (code && code.data) return code.data;

// Pass 3: Adaptive High-Contrast Binarization
const data = imageData.data;
const binarized = new Uint8ClampedArray(data.length);

// Convert to Grayscale & Contrast Thresholding
for (let i = 0; i < data.length; i += 4) {
const avg = (data[i] + data[i + 1] + data[i + 2]) / 3;
const v = avg < 128 ? 0 : 255; // Threshold binary grid cell
binarized[i] = v;
binarized[i + 1] = v;
binarized[i + 2] = v;
binarized[i + 3] = 255;
}

code = jsQR(binarized, imageData.width, imageData.height, {
inversionAttempts: "attemptBoth",
});
return code ? code.data : null;
}
